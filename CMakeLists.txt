cmake_minimum_required(VERSION 3.18)
project(voxelmill LANGUAGES CXX)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
# 3.0.1 caches a dangling Python thread state when first imported by Qt workers.
find_package(pybind11 3.0.4 CONFIG REQUIRED)
set(NATIVE_SOURCES native/module.cpp native/mesh.cpp native/raster.cpp native/voxel.cpp native/goo.cpp native/ctb.cpp native/distance.cpp native/runs.cpp native/edt.cpp)
include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  enable_language(CUDA)
  list(APPEND NATIVE_SOURCES native/cuda_bind.cpp native/cuda_morphology.cu)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/native/intersections.cpp")
  list(APPEND NATIVE_SOURCES native/intersections.cpp)
endif()
pybind11_add_module(_native ${NATIVE_SOURCES})
target_compile_features(_native PRIVATE cxx_std_17)
if(CMAKE_CUDA_COMPILER)
  target_compile_definitions(_native PRIVATE VOXELMILL_CUDA=1)
  set_target_properties(_native PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
find_package(TBB QUIET)
if(TBB_FOUND)
 target_link_libraries(_native PRIVATE TBB::tbb)
 target_compile_definitions(_native PRIVATE VOXELMILL_TBB=1)
 message(STATUS "voxelmill: oneTBB found; native loops use tbb::parallel_for")
else()
 # native/parallel.hpp falls back to std::thread, so kernels stay parallel.
 find_package(Threads REQUIRED)
 target_link_libraries(_native PRIVATE Threads::Threads)
 message(STATUS "voxelmill: oneTBB not found; native loops use the std::thread fallback")
endif()
install(TARGETS _native LIBRARY DESTINATION voxelmill)
